Assignment Tracker Portal (Google Apps Script + Google Sheets)

A teacher-friendly setup guide to build a secure web portal where each student sees only their own record (grades / assignment status) pulled from a Google Sheet. Students authenticate with their Google Workspace for Education account (e.g. @students.edu.sg).

Option A: Apps Script Web App (HTML Service) Email-based validation No CORS issues Works well for school domains

1) System overview

This portal works like a “self-service report” page. The Sheet is your database. The Web App is your secure gateway. When a student opens the portal:

  1. Google confirms they are signed in to your school domain account.
  2. The Apps Script server reads the logged-in user’s email.
  3. It finds the matching row in your Sheet (Email = key).
  4. It returns only that row to the student UI.
If you deploy as Execute as “User accessing the web app”, students must have access to the Sheet. To avoid permission problems, deploy as Execute as “Me”.

2) Spreadsheet setup

Create a Google Sheet tab named Grades (you can rename it) with headers like:

Email | Class | Gender | Name | Assignment 1 | Assignment 2 | Overall | Remarks

Only the first four fields are “always shown” in the student profile. Email is the validation key. Everything after that is displayed as the “Assignments / Grades” list.

Tip: Keep header names consistent. Use exactly: Email, Class, Gender, Name.
Example: Email as the key, with subsequent assignment columns.

3) Apps Script project

Go to script.google.com → New project. Create 2 files: Code.gs and Index.html.

Reminder: In Code.gs, set your spreadsheet and tab:
const SPREADSHEET_ID = "PASTE_YOUR_SPREADSHEET_ID_HERE";
const SHEET_NAME = "Grades"; // change to your tab name

// ====================== CONFIG ======================
const SPREADSHEET_ID = "PASTE_YOUR_SPREADSHEET_ID_HERE";
const SHEET_NAME = "Grades"; // change to your tab name
// ====================================================

function doGet(e) {
  // Template lets us pass the deployment URL into the HTML (useful for account switch/logout links)
  const t = HtmlService.createTemplateFromFile("Index");
  t.webAppUrl = ScriptApp.getService().getUrl();

  return t.evaluate()
    .setTitle("Assignment Tracker")
    // Allow embedding in an iframe (e.g., LMS, Google Sites, WordPress).
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/**
 * Returns the record for the logged-in student.
 * Email is used as the unique identifier / validation key.
 */
function getMyGrades() {
  // For Google Workspace domains, ActiveUser typically returns the student's email.
  // EffectiveUser is a fallback in some environments.
  const email =
    Session.getActiveUser().getEmail() ||
    Session.getEffectiveUser().getEmail();

  if (!email) {
    return {
      ok: false,
      error:
        "Could not detect your email. Please open this portal while signed in to your school account."
    };
  }

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) return { ok: false, error: `Sheet tab '${SHEET_NAME}' not found.` };

  const values = sh.getDataRange().getValues();
  if (!values || values.length < 2) return { ok: false, error: "No data found in sheet." };

  const headers = values[0].map(h => String(h || "").trim());
  const norm = v => String(v || "").trim().toLowerCase();

  // Find Email column: exact "Email" preferred, else any header containing "email"
  let emailCol = headers.findIndex(h => norm(h) === "email");
  if (emailCol === -1) emailCol = headers.findIndex(h => norm(h).includes("email"));
  if (emailCol === -1) return { ok: false, error: "Could not find an Email column header (must contain 'Email')." };

  // Find the student row by email (case-insensitive)
  const target = norm(email);
  let rowIndex = -1;
  for (let r = 1; r < values.length; r++) {
    if (norm(values[r][emailCol]) === target) { rowIndex = r; break; }
  }
  if (rowIndex === -1) return { ok: false, error: `No record found for ${email}.` };

  const row = values[rowIndex];
  const colIndex = (headerName) => headers.findIndex(h => norm(h) === norm(headerName));

  // Always-show fields (if present)
  const idxIcon   = colIndex("ICON");   // optional
  const idxClass  = colIndex("Class");
  const idxGender = colIndex("Gender");
  const idxName   = colIndex("Name");

  const profile = {
    ...(idxIcon !== -1 ? { ICON: row[idxIcon] } : {}),
    Email:  row[emailCol],
    Class:  idxClass  !== -1 ? row[idxClass]  : "",
    Gender: idxGender !== -1 ? row[idxGender] : "",
    Name:   idxName   !== -1 ? row[idxName]   : ""
  };

  // All other columns become the "Assignments / Grades" list
  const exclude = new Set([idxIcon, emailCol, idxClass, idxGender, idxName].filter(i => i !== -1));
  const grades = [];
  headers.forEach((h, i) => {
    if (!h) return;
    if (exclude.has(i)) return;
    grades.push({ field: h, value: row[i] });
  });

  return { ok: true, email, profile, grades };
}

<body>
  <div class="wrap">
    <div class="card">
      <h1>Assignment Tracker</h1>
      <p class="muted">You’ll see only the record linked to your signed-in Google account.</p>

      <div class="row">
        <button id="loadBtn" class="btn" onclick="loadGrades()">Load my record</button>
        <button class="btn secondary danger" onclick="googleLogout()" type="button">
          Log out of Google
        </button>

        <span id="who" class="pill" style="display:none;"></span>
      </div>

  <script>
    const WEB_APP_URL = "<?= webAppUrl ?>";

    // Logs out in that browser session (may sign out of all Google tabs)
    function googleLogout(){
      const url =
        "https://accounts.google.com/Logout?continue=" +
        encodeURIComponent(WEB_APP_URL);
      window.open(url, "_blank", "noopener");
    }

    function setMsg(text, isError=false){
      const el = document.getElementById("msg");
      el.textContent = text || "";
      el.style.color = isError ? "#b91c1c" : "#0f172a";
    }

    function escapeHtml(s){
      s = String(s ?? "");
      return s.replace(/[&<>"']/g, c => (
        {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[c]
      ));
    }

    function renderProfile(profile){
      const el = document.getElementById("profile");
      if (!profile) { el.innerHTML = ""; return; }

      const rows = [];
      if (profile.ICON !== undefined && String(profile.ICON).trim() !== "") {
        rows.push(["ICON", profile.ICON]);
      }
      rows.push(["Email", profile.Email]);
      rows.push(["Class", profile.Class]);
      rows.push(["Gender", profile.Gender]);
      rows.push(["Name", profile.Name]);

      el.innerHTML = `
        <h3>Student Profile</h3>
        <table aria-label="Student Profile">
          ${rows.map(([k,v]) => `
            <tr><th>${escapeHtml(k)}</th><td>${escapeHtml(v ?? "")}</td></tr>
          `).join("")}
        </table>
        <div class="divider"></div>
      `;
    }

    function renderGrades(grades){
      const el = document.getElementById("grades");
      if (!grades || !grades.length) {
        el.innerHTML = `<p class="muted" style="margin-top:6px;">No assignment fields found.</p>`;
        return;
      }

      el.innerHTML = `
        <h3>Assignments / Grades</h3>
        <table aria-label="Assignments / Grades">
          ${grades.map(item => `
            <tr><th>${escapeHtml(item.field)}</th><td>${escapeHtml(item.value ?? "")}</td></tr>
          `).join("")}
        </table>
      `;
    }

    function loadGrades(){
      const btn = document.getElementById("loadBtn");
      btn.disabled = true;

      setMsg("Loading…");
      document.getElementById("profile").innerHTML = "";
      document.getElementById("grades").innerHTML = "";
      document.getElementById("who").style.display = "none";

      google.script.run
        .withSuccessHandler(res => {
          btn.disabled = false;

          if (!res || !res.ok) {
            setMsg(res?.error || "Unknown error.", true);
            return;
          }

          setMsg("");
          const who = document.getElementById("who");
          who.textContent = res.email;
          who.style.display = "inline-block";

          renderProfile(res.profile);
          renderGrades(res.grades);
        })
        .withFailureHandler(err => {
          btn.disabled = false;
          setMsg(err?.message || "Failed to load record.", true);
        })
        .getMyGrades();
    }
  </script>

4) Deploy as a Web App

  1. Apps Script → Deploy → New deployment → Select Web app.
  2. Execute as: Me
  3. Who has access: Anyone within your domain (e.g., students.edu.sg)
  4. Deploy → Copy the Web App URL.
Share the Web App URL with your students to access the app. They will authenticate with their school Google accounts and see only their own record.
